<script>
document.addEventListener('DOMContentLoaded', () => {
    const searchInput = document.getElementById('exerciseSearch');
    const selectedFilter = document.getElementById('showOnlySelected');
    const cards = document.querySelectorAll('.exercise-card');
    const noResultsMsg = document.getElementById('noResultsMsg');
    const matchCount = document.getElementById('matchCount');

    function updateCardHighlight(card, isChecked) {
        if (isChecked) {
            card.classList.add('border-l-emerald-500', 'bg-emerald-950/10');
        } else {
            card.classList.remove('border-l-emerald-500', 'bg-emerald-950/10');
        }
    }

    function filterExercises() {
        const query = searchInput.value.trim().toLowerCase();
        const showSelectedOnly = selectedFilter.checked;
        let visibleCount = 0;

        cards.forEach(card => {
            const name = card.dataset.name || '';
            const instructions = card.dataset.instructions || '';
            const muscles = card.dataset.muscles || '';
            const checkbox = card.querySelector('.exercise-checkbox');
            const isChecked = checkbox.checked;

            updateCardHighlight(card, isChecked);

            const matchesSearch = !query || name.includes(query) || instructions.includes(query) || muscles.includes(query);
            const matchesSelectedFilter = !showSelectedOnly || isChecked;

            if (matchesSearch && matchesSelectedFilter) {
                card.classList.remove('hidden');
                visibleCount++;
            } else {
                card.classList.add('hidden');
            }
        });

        if (visibleCount === 0) {
            noResultsMsg.classList.remove('hidden');
        } else {
            noResultsMsg.classList.add('hidden');
        }

        matchCount.textContent = `${visibleCount} of ${cards.length}`;
    }

    searchInput.addEventListener('input', filterExercises);
    selectedFilter.addEventListener('change', filterExercises);
    
    cards.forEach(card => {
        const cb = card.querySelector('.exercise-checkbox');
        cb.addEventListener('change', filterExercises);
    });

    filterExercises();
});
</script>
{% endblock %}

    <!-- Audio Engine Scripts -->
    <script>
        window.CMFitSoundConfig = {{ sound_config | tojson | safe if sound_config is defined else '{}' }};
        window.audioCache = {};
        let audioCtx = null;
        window.soundBuffers = {};

        function getSoundDetails(type) {
            if (!window.CMFitSoundConfig) return { enabled: false, url: null };
            const cfg = window.CMFitSoundConfig;
            if (type === 'start') return { enabled: !!cfg.start_enabled, url: cfg.start_url };
            if (type === 'notice') return { enabled: !!cfg.notice_enabled, url: cfg.notice_url };
            if (type === 'rest_end') return { enabled: !!cfg.rest_end_enabled, url: cfg.rest_end_url };
            if (type === 'end') return { enabled: !!cfg.end_enabled, url: cfg.end_url };
            return { enabled: false, url: null };
        }

        function initAudioEngine() {
            if (!audioCtx) {
                const AudioContextClass = window.AudioContext || window.webkitAudioContext;
                if (AudioContextClass) audioCtx = new AudioContextClass();
            }
            if (audioCtx && audioCtx.state === 'suspended') audioCtx.resume();

            ['start', 'notice', 'rest_end', 'end'].forEach(type => {
                const details = getSoundDetails(type);
                if (details.enabled && details.url && !window.audioCache[type]) {
                    const a = new Audio(details.url);
                    a.preload = 'auto';
                    window.audioCache[type] = a;
                    if (audioCtx) {
                        fetch(details.url)
                            .then(res => res.arrayBuffer())
                            .then(buf => audioCtx.decodeAudioData(buf))
                            .then(decoded => { window.soundBuffers[type] = decoded; })
                            .catch(e => console.warn('WebAudio fallback for', type, e));
                    }
                }
            });
        }

        function unlockAudio() {
            initAudioEngine();
            Object.keys(window.audioCache).forEach(k => {
                const sound = window.audioCache[k];
                sound.muted = true;
                sound.play().then(() => {
                    sound.pause();
                    sound.currentTime = 0;
                    sound.muted = false;
                }).catch(() => { sound.muted = false; });
            });
            document.removeEventListener('touchstart', unlockAudio);
            document.removeEventListener('click', unlockAudio);
        }

        document.addEventListener('touchstart', unlockAudio, { once: true, passive: true });
        document.addEventListener('click', unlockAudio, { once: true, passive: true });

        window.playCMFitSound = function(type) {
            const details = getSoundDetails(type);
            if (!details.enabled || !details.url) return;

            if (audioCtx && window.soundBuffers[type]) {
                try {
                    if (audioCtx.state === 'suspended') audioCtx.resume();
                    const source = audioCtx.createBufferSource();
                    source.buffer = window.soundBuffers[type];
                    source.connect(audioCtx.destination);
                    source.start(0);
                    return;
                } catch(e) {}
            }

            let audio = window.audioCache[type];
            if (!audio) {
                audio = new Audio(details.url);
                window.audioCache[type] = audio;
            }
            audio.currentTime = 0;
            audio.play().catch(e => console.error('Audio play error:', e));
        };
    </script>

<script>
document.addEventListener('DOMContentLoaded', () => {
    const searchInput = document.getElementById('exerciseSearch');
    const clearBtn = document.getElementById('clearSearch');
    const cards = document.querySelectorAll('.exercise-card');
    const noResultsMsg = document.getElementById('noResultsMsg');
    const counterBadge = document.getElementById('exerciseCounter');
    const totalCount = cards.length;

    function filterExercises() {
        const query = searchInput.value.trim().toLowerCase();
        let visibleCount = 0;

        if (query.length > 0) {
            clearBtn.classList.remove('hidden');
        } else {
            clearBtn.classList.add('hidden');
        }

        cards.forEach(card => {
            const name = card.dataset.name || '';
            const instructions = card.dataset.instructions || '';
            const muscles = card.dataset.muscles || '';
            const matches = !query || name.includes(query) || instructions.includes(query) || muscles.includes(query);

            if (matches) {
                card.classList.remove('hidden');
                visibleCount++;
            } else {
                card.classList.add('hidden');
            }
        });

        if (counterBadge) {
            counterBadge.textContent = visibleCount;
        }

        if (visibleCount === 0 && totalCount > 0) {
            noResultsMsg.classList.remove('hidden');
        } else {
            noResultsMsg.classList.add('hidden');
        }
    }

    searchInput.addEventListener('input', filterExercises);

    clearBtn.addEventListener('click', () => {
        searchInput.value = '';
        filterExercises();
        searchInput.focus();
    });
});
</script>
{% endblock %}

<!-- Drag-and-Drop Script -->
<script>
document.addEventListener('DOMContentLoaded', () => {
    const list = document.getElementById('exercise-list');
    let draggedItem = null;

    list.querySelectorAll('.reorder-item').forEach(item => {
        item.setAttribute('draggable', 'true');

        item.addEventListener('dragstart', (e) => {
            draggedItem = item;
            item.classList.add('opacity-40', 'scale-[0.98]');
        });

        item.addEventListener('dragend', () => {
            draggedItem.classList.remove('opacity-40', 'scale-[0.98]');
            draggedItem = null;
            saveOrder();
        });

        item.addEventListener('dragover', (e) => {
            e.preventDefault();
            const afterElement = getDragAfterElement(list, e.clientY);
            if (afterElement == null) {
                list.appendChild(draggedItem);
            } else {
                list.insertBefore(draggedItem, afterElement);
            }
        });
    });

    function getDragAfterElement(container, y) {
        const draggableElements = [...container.querySelectorAll('.reorder-item:not(.opacity-40)')];
        return draggableElements.reduce((closest, child) => {
            const box = child.getBoundingClientRect();
            const offset = y - box.top - box.height / 2;
            if (offset < 0 && offset > closest.offset) {
                return { offset: offset, element: child };
            } else {
                return closest;
            }
        }, { offset: Number.NEGATIVE_INFINITY }).element;
    }

    function saveOrder() {
        const orderData = [];
        list.querySelectorAll('.reorder-item').forEach((el, index) => {
            orderData.push({
                id: parseInt(el.getAttribute('data-we-id')),
                order: index
            });
        });

        fetch("{{ url_for('reorder_workout_exercises', workout_id=workout.id) }}", {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ order: orderData })
        });
    }
});
</script>
